### 📱 36. PAGBABAGO SA GITHUB GAMIT ANG TERMUX

> ✅ **ITATYPE MO LANG:** `./update-github`
> ✅ **HIHINGIN LAMANG NG ISANG BESES:** Username, Repository, Token — HINDI NA ULITIN!
> ✅ **TATLONG PILIAN:**
>   - 📂 **1 — BUONG FILE** → Ilagay ang path → I-paste → `Ctrl + D` → I-send
>   - ✨ **2 — PATCH LANG** → Ilagay ang path → Hanapin → Palitan → Awtomatiko → I-send
>   - ❌ **3 — Isara** → Tapusin na
> ✅ **AWTOMATIKONG I-SA-SAVE SA GITHUB → BUBUO NG APK!**

#### 📜 BUONG KODIGO NG `update-github`:
```bash
#!/data/data/com.termux/files/usr/bin/bash
clear
echo "=========================================="
echo "🔧 BUONG FILE O PATCH LANG"
echo "=========================================="
echo ""

CONFIG_FILE="$HOME/.github_config"

if [ -f "$CONFIG_FILE" ]; then
  source "$CONFIG_FILE"
else
  echo "⚠️ UNANG BESES — ILAGAY ANG IMPORMASYON:"
  read -p "👤 Username: " USERNAME
  read -p "📁 Repository: " REPO
  read -s -p "🔑 Token: " TOKEN
  echo ""
  [ -z "$USERNAME" ] || [ -z "$REPO" ] || [ -z "$TOKEN" ] && { echo "❌ HINDI PWEDENG WALANG LAMAN!"; exit 1; }
  cat > "$CONFIG_FILE" << EOF
USERNAME=$USERNAME
REPO=$REPO
TOKEN=$TOKEN
EOF
  chmod 600 "$CONFIG_FILE"
  echo "✅ NAI-SAVE NA! HINDI NA ULITIN!"
  echo ""
fi

cd ~/$REPO 2>/dev/null || { mkdir -p ~/$REPO; cd ~/$REPO; git init; }
git remote set-url origin "https://${TOKEN}@github.com/${USERNAME}/${REPO}.git"

echo "✅ ANO ANG GAGAWIN?"
echo ""
echo "📂 1 — BUONG FILE: Palitan lahat ng laman"
echo "✨ 2 — PATCH LANG: Hanapin at palitan lang"
echo "❌ 3 — Isara"
echo ""
read -p "Piliin (1, 2 o 3): " PILI

case $PILI in
  1)
    echo ""
    read -p "📄 Path ng file: " FILE
    [ -z "$FILE" ] && { echo "❌ WALANG FILE!"; exit 1; }
    echo "✅ IPASTE ANG BAGONG LAMAN → Ctrl+D kapag tapos:"
    echo "---------------------------------------------------"
    mkdir -p "$(dirname "$FILE")"
    cat > "$FILE"
    echo "---------------------------------------------------"
    echo "✅ NA-ISULAT NA!"
    ;;

  2)
    echo ""
    echo "✨ === PATCH LANG ==="
    read -p "📄 Path ng file: " FILE
    [ -z "$FILE" ] || [ ! -f "$FILE" ] && { echo "❌ HINDI MAHANAP ANG FILE!"; exit 1; }
    read -p "🔍 Hanapin ang: " HANAP
    read -p "✏️  Palitan ng: " PALITAN
    sed -i "s/$HANAP/$PALITAN/g" "$FILE"
    echo "✅ NA-PATCH NA! '$HANAP' → '$PALITAN'"
    ;;

  3)
    echo "✅ TAPOS NA!"
    exit 0
    ;;

  *)
    echo "❌ MALING PINILI!"
    exit 1
    ;;
esac

echo ""
read -p "💬 Mensahe: " MENSAHE
git add .
git commit -m "$MENSAHE"
git push -u origin main

if [ $? -eq 0 ]; then
  echo "✅✅✅ NA-UPLOAD NA!"
else
  echo "⚠️ Hihilahin muna..."
  git pull -X theirs
  git push -u origin main
  [ $? -eq 0 ] && echo "✅✅✅ TAGUMPAY NA!" || echo "❌ Kailangan ayusin manu-mano!"
fi
